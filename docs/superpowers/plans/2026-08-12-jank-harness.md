# Jank / Frame-Drop Test Harness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an interactive, visible jank/frame-drop test suite to `playground/harmony` covering structural (JSON story) and temporal (ArkTS-driven) jank, with a real-time FPS overlay.

**Architecture:** Plan B from the design spec — structural jank becomes new `A2UI Show/Jank` JSON stories (auto-appear in the `AGenUIDemoPage` menu) with a shared `FpsOverlay` toggle added to that page; temporal jank lives in a new `JankTestPage` that mounts a visible `AGenUIContainer` on a small baseline surface and runs ArkTS bad-behavior drivers against it, reusing the same `FpsOverlay`. FPS is measured in-app via `@ohos.graphics.displaySync` `on('frame')` callbacks.

**Tech Stack:** HarmonyOS ArkTS (`.ets`), `@agenui/agenui` SDK (`SurfaceManager` / `AGenUIContainer`), `@ohos.graphics.displaySync`, hypium (`@ohos/hypium`) for tests, Python 3 for the story generator script.

```python
#!/usr/bin/env python3
"""Generate A2UI Show/Jank structural-jank stories into resource + harmony rawfile."""
import json
import os

REPO = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
RESOURCE_BASE = os.path.join(REPO, "playground", "resource", "stories", "A2UI Show", "Jank")
RAWFILE_BASE = os.path.join(
    REPO, "playground", "harmony", "entry", "src", "main", "resources",
    "rawfile", "stories", "A2UI Show", "Jank",
)

def write(name: str, payload: dict) -> None:
    for base in (RESOURCE_BASE, RAWFILE_BASE):
        d = os.path.join(base, name)
        os.makedirs(d, exist_ok=True)
        with open(os.path.join(d, "updateComponents.json"), "w", encoding="utf-8") as f:
            json.dump(payload, f, ensure_ascii=False, indent=2)

def massive_tree() -> dict:
    children = [f"t{i}" for i in range(500)]
    components = [{"id": "root", "component": "Column", "align": "stretch", "children": children}]
    components += [{"id": f"t{i}", "component": "Text", "text": f"Item {i}", "variant": "body"} for i in range(500)]
    return {"version": "v0.9", "updateComponents": {"surfaceId": "jank-massive", "components": components}}

def deep_nesting() -> dict:
    # 30 levels of nested Columns; level i references level i+1 (as "nest{i+1}")
    # plus 3 Text siblings. Root is "nest0" (id "root" alias handled by surface).
    ids_text = [f"d{i}-t{j}" for i in range(30) for j in range(3)]
    comps = []
    for i in range(30):
        child_ids = [f"d{i}-t0", f"d{i}-t1", f"d{i}-t2"]
        if i < 29:
            child_ids.append(f"nest{i+1}")
        comps.append({"id": "root" if i == 0 else f"nest{i}", "component": "Column", "children": child_ids})
    comps += [{"id": t, "component": "Text", "text": "deep", "variant": "body"} for t in ids_text]
    return {"version": "v0.9", "updateComponents": {"surfaceId": "jank-deep", "components": comps}}

def overdraw() -> dict:
    layers = [f"s{i}" for i in range(20)]
    components = [{"id": "root", "component": "Stack", "children": layers}]
    components += [{"id": f"s{i}", "component": "Column", "backgroundColor": "rgba(255,0,0,0.2)", "width": "100%", "height": "100%"} for i in range(20)]
    components.append({"id": "center", "component": "Text", "text": "OVERDRAW", "variant": "h1"})
    components[0]["children"].append("center")
    return {"version": "v0.9", "updateComponents": {"surfaceId": "jank-overdraw", "components": components}}

def huge_payload() -> dict:
    big = "x" * 200000
    components = [
        {"id": "root", "component": "Column", "children": ["big"]},
        {"id": "big", "component": "Text", "text": big, "variant": "body"},
    ]
    return {"version": "v0.9", "updateComponents": {"surfaceId": "jank-huge", "components": components}}

def main() -> None:
    write("MassiveTree", massive_tree())
    write("DeepNesting", deep_nesting())
    write("Overdraw", overdraw())
    write("HugePayload", huge_payload())
    print("generated 4 jank stories into resource + rawfile")

if __name__ == "__main__":
    main()
```

```typescript
import { hilog } from '@kit.PerformanceAnalysisKit';

const TAG = 'JankScheduler';
const DOMAIN = 0x0000;

/**
 * Manages setInterval/setTimeout handles for jank scenarios.
 *
 * Pure round-cap logic is static and unit-tested; handle accounting is
 * thin so all running loops can be stopped from one place.
 */
export class JankLoopScheduler {
  /** Hard ceiling on scenario iterations (backstop for forgetting to press Stop). */
  static readonly MAX_ROUNDS: number = 2000;

  /**
   * True when `round` has reached/exceeded `maxRounds`. A `maxRounds` of 0
   * means "no cap" (returns false). Used by each scenario's tick to decide
   * whether to keep looping.
   */
  static isRoundCapped(round: number, maxRounds: number): boolean {
    return maxRounds > 0 && round >= maxRounds;
  }

  private handles: number[] = [];

  /** Register a timer handle so stopAll() can clear it. Returns the handle. */
  register(handle: number): number {
    this.handles.push(handle);
    return handle;
  }

  /** True if at least one registered handle is active. */
  hasActive(): boolean {
    return this.handles.length > 0;
  }

  /** Clear all registered handles and return how many were cleared. Idempotent. */
  stopAll(): number {
    const n = this.handles.length;
    for (let i = 0; i < this.handles.length; i++) {
      const h = this.handles[i];
      clearInterval(h);
      clearTimeout(h);
    }
    this.handles = [];
    if (n > 0) {
      hilog.info(DOMAIN, TAG, 'stopped %{public}d handle(s)', n);
    }
    return n;
  }
}
```

```typescript
import displaySync from '@ohos.graphics.displaySync';
import { hilog } from '@kit.PerformanceAnalysisKit';

const TAG = 'FpsOverlay';
const DOMAIN = 0x0000;

/**
 * Semi-transparent overlay that measures effective main-thread frame rate
 * using displaySync.on('frame'). When the main thread is blocked, the frame
 * callback itself is delayed, so callback spacing reflects real jank.
 *
 * Drop-in usable from any page: set @Prop scenario/errorText, mount in a
 * Stack. Cleans up its displaySync + timer in aboutToDisappear.
 */
@Component
export struct FpsOverlay {
  @Prop scenario: string = '';
  @Prop errorText: string = '';
  @State fps: number = 0;
  @State avgFrameMs: number = 0;
  @State dropped: number = 0;

  private ds: displaySync.DisplaySync | null = null;
  private frameTimes: number[] = [];
  private aggregateTimer: number = -1;
  private lastFrameWall: number = -1;
  private droppedCount: number = 0;
  private readonly expectedIntervalMs: number = 1000 / 60;

  aboutToAppear() {
    this.ds = displaySync.create();
    this.ds.setExpectedFrameRateRange({ min: 0, max: 120, expected: 60 });
    this.ds.on('frame', () => {
      const now = Date.now();
      if (this.lastFrameWall >= 0) {
        const delta = now - this.lastFrameWall;
        if (delta > this.expectedIntervalMs * 1.5) {
          this.droppedCount++;
        }
      }
      this.lastFrameWall = now;
      this.frameTimes.push(now);
      if (this.frameTimes.length > 240) {
        this.frameTimes.shift();
      }
    });
    this.ds.start();
    this.aggregateTimer = setInterval(() => this.aggregate(), 500);
    hilog.info(DOMAIN, TAG, 'displaySync started');
  }

  private aggregate() {
    if (this.frameTimes.length === 0) {
      this.fps = 0;
      this.avgFrameMs = 0;
      return;
    }
    const now = Date.now();
    const oneSecondAgo = now - 1000;
    let recentStart = 0;
    for (let i = this.frameTimes.length - 1; i >= 0; i--) {
      if (this.frameTimes[i] < oneSecondAgo) {
        recentStart = i + 1;
        break;
      }
    }
    const recent = this.frameTimes.slice(recentStart);
    const count = recent.length;
    if (count === 0) {
      this.fps = 0;
      return;
    }
    const span = count > 1 ? (recent[count - 1] - recent[0]) : 0;
    let fps = span > 0 ? Math.round(((count - 1) / span) * 1000) : count;
    if (fps > 120) { fps = 120; }
    this.fps = fps;
    let sumDelta = 0;
    let deltaCount = 0;
    for (let i = 1; i < recent.length; i++) {
      sumDelta += recent[i] - recent[i - 1];
      deltaCount++;
    }
    this.avgFrameMs = deltaCount > 0 ? Math.round((sumDelta / deltaCount) * 10) / 10 : 0;
    this.dropped = this.droppedCount;
  }

  aboutToDisappear() {
    if (this.aggregateTimer !== -1) {
      clearInterval(this.aggregateTimer);
      this.aggregateTimer = -1;
    }
    if (this.ds) {
      this.ds.stop();
      this.ds.off('frame');
      this.ds = null;
      hilog.info(DOMAIN, TAG, 'displaySync stopped');
    }
  }

  build() {
    Column() {
      Text(this.scenario || 'IDLE')
        .fontSize(11).fontColor('#FFFFFF').fontFamily('monospace').margin({ bottom: 2 })
      Text(`FPS:${this.fps}`)
        .fontSize(13)
        .fontColor(this.fps >= 55 ? '#4CFF4C' : (this.fps >= 30 ? '#FFCC33' : '#FF6666'))
        .fontFamily('monospace').margin({ bottom: 2 })
      Text(`frame:${this.avgFrameMs}ms`)
        .fontSize(11).fontColor('#FFFFFF').fontFamily('monospace').margin({ bottom: 2 })
      Text(`dropped:${this.dropped}`)
        .fontSize(11).fontColor(this.dropped > 0 ? '#FF6666' : '#FFFFFF').fontFamily('monospace').margin({ bottom: 2 })
      if (this.errorText) {
        Text(this.errorText).fontSize(10).fontColor('#FF6666').fontFamily('monospace').maxLines(2)
      }
    }
    .padding({ left: 12, right: 12, top: 10, bottom: 10 })
    .borderRadius(12)
    .backgroundColor('rgba(0, 0, 0, 0.7)')
    .margin({ left: 16, top: 60 })
    .alignItems(HorizontalAlign.Start)
  }
}
```

```typescript
import { SurfaceManager } from '@agenui/agenui';
import { common } from '@kit.AbilityKit';
import { hilog } from '@kit.PerformanceAnalysisKit';
import { JankLoopScheduler } from './JankLoopScheduler';
import {
  JANK_BASELINE_SURFACE_ID,
  JANK_BASELINE_JSON,
  JANK_BASELINE_CHUNK
} from './jankStories';

const TAG = 'JankEngine';
const DOMAIN = 0x0000;
const CATALOG = 'https://a2ui.org/specification/v0_9/basic_catalog.json';

export type JankRunState = (name: string, running: boolean, error: string | null) => void;
export type JankAnimTick = (pos: number) => void;

/**
 * Runs temporal-jank scenarios against a visible baseline surface held by
 * JankTestPage. Each scenario schedules work via setInterval and is
 * registered with a JankLoopScheduler so stop() can abort everything.
 */
export class JankScenarioEngine {
  static readonly SCENARIO_NAMES: string[] = [
    'MainThreadBlock', 'StateChurn', 'TinyChunkStream',
    'SetTimeoutAnimation', 'SurfaceThrash', 'MemoryChurn'
  ];

  static isKnown(name: string): boolean {
    return JankScenarioEngine.SCENARIO_NAMES.includes(name);
  }

  private context: common.UIAbilityContext;
  private sm: SurfaceManager;
  private scheduler: JankLoopScheduler = new JankLoopScheduler();
  private onState: JankRunState;
  private onAnimTick: JankAnimTick;
  private animPos: number = 0;

  constructor(
    context: common.UIAbilityContext,
    sm: SurfaceManager,
    onState: JankRunState,
    onAnimTick: JankAnimTick
  ) {
    this.context = context;
    this.sm = sm;
    this.onState = onState;
    this.onAnimTick = onAnimTick;
  }

  /** Create + populate the baseline surface so there is something visible to jank. */
  loadBaseline(): void {
    this.sm.beginTextStream();
    this.sm.receiveTextChunk(
      `{"version":"v0.9","createSurface":{"surfaceId":"${JANK_BASELINE_SURFACE_ID}","catalogId":"${CATALOG}"}}`
    );
    this.sm.receiveTextChunk(JANK_BASELINE_JSON);
    this.sm.endTextStream();
    hilog.info(DOMAIN, TAG, 'baseline surface loaded');
  }

  /** Run a scenario by name. Unknown names are a no-op (logged). */
  run(name: string): void {
    if (!JankScenarioEngine.isKnown(name)) {
      hilog.warn(DOMAIN, TAG, 'unknown scenario: %{public}s', name);
      return;
    }
    this.stop();
    this.animPos = 0;
    this.onState(name, true, null);
    try {
      switch (name) {
        case 'MainThreadBlock': this.mainThreadBlock(); break;
        case 'StateChurn': this.stateChurn(); break;
        case 'TinyChunkStream': this.tinyChunkStream(); break;
        case 'SetTimeoutAnimation': this.setTimeoutAnimation(); break;
        case 'SurfaceThrash': this.surfaceThrash(); break;
        case 'MemoryChurn': this.memoryChurn(); break;
      }
    } catch (e) {
      this.onState(name, false, (e as Error).message);
      hilog.error(DOMAIN, TAG, 'scenario crashed: %{public}s', String(e));
    }
  }

  /** Stop all running loops. Idempotent. */
  stop(): void {
    this.scheduler.stopAll();
    this.onState('', false, null);
  }

  private mainThreadBlock(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      const end = Date.now() + 50;
      while (Date.now() < end) { /* busy-wait 50ms on main thread */ }
      round++;
    };
    const h = setInterval(tick, 100);
    this.scheduler.register(h);
  }

  private stateChurn(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      this.sm.syncState(JANK_BASELINE_SURFACE_ID, 'block', JSON.stringify({ t: round }));
      round++;
    };
    const h = setInterval(tick, 5);
    this.scheduler.register(h);
  }

  private tinyChunkStream(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      this.sm.beginTextStream();
      this.sm.receiveTextChunk(JANK_BASELINE_CHUNK);
      this.sm.endTextStream();
      round++;
    };
    const h = setInterval(tick, 30);
    this.scheduler.register(h);
  }

  private setTimeoutAnimation(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      this.animPos = (this.animPos + 4) % 200;
      this.onAnimTick(this.animPos);
      round++;
    };
    const h = setInterval(tick, 16);
    this.scheduler.register(h);
  }

  private surfaceThrash(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      this.sm.beginTextStream();
      this.sm.receiveTextChunk(
        `{"version":"v0.9","deleteSurface":{"surfaceId":"${JANK_BASELINE_SURFACE_ID}"}}`
      );
      this.sm.receiveTextChunk(
        `{"version":"v0.9","createSurface":{"surfaceId":"${JANK_BASELINE_SURFACE_ID}","catalogId":"${CATALOG}"}}`
      );
      this.sm.receiveTextChunk(JANK_BASELINE_JSON);
      this.sm.endTextStream();
      round++;
    };
    const h = setInterval(tick, 50);
    this.scheduler.register(h);
  }

  private memoryChurn(): void {
    let round = 0;
    const tick = (): void => {
      if (JankLoopScheduler.isRoundCapped(round, JankLoopScheduler.MAX_ROUNDS)) { this.stop(); return; }
      // Allocate ~5MB then drop the reference, forcing periodic GC pauses.
      const buf: number[] = new Array(655360);
      buf[0] = round;
      round++;
    };
    const h = setInterval(tick, 16);
    this.scheduler.register(h);
  }
}
```

```typescript
import { AGenUIContainer, SurfaceManager, ISurfaceManagerListener, Surface } from '@agenui/agenui';
import { common } from '@kit.AbilityKit';
import { router } from '@kit.ArkUI';
import { hilog } from '@kit.PerformanceAnalysisKit';
import { FpsOverlay } from '../components/FpsOverlay';
import { JankScenarioEngine } from './JankScenarioEngine';

const TAG = 'JankTestPage';
const DOMAIN = 0x0000;
const COLORS = ['#6200EE', '#3700B3', '#4D95E5', '#BB86FC', '#03DAC5', '#FF6666'];

@Entry
@Component
struct JankTestPage {
  @State surfaceId: string = '';
  @State surfaceVersion: number = 0;
  @State activeScenario: string = '';
  @State isRunning: boolean = false;
  @State errorText: string = '';
  @State animPos: number = 0;
  @State smoothRunning: boolean = false;

  private sm: SurfaceManager | null = null;
  private engine: JankScenarioEngine | null = null;

  aboutToAppear() {
    const context = getContext(this) as common.UIAbilityContext;
    this.sm = new SurfaceManager(context);
    this.sm.addListener({
      onCreateSurface: (s: Surface) => {
        this.surfaceVersion++;
        this.surfaceId = s.surfaceId;
        hilog.info(DOMAIN, TAG, 'surface created: %{public}s', s.surfaceId);
      },
      onDeleteSurface: (s: Surface) => {
        if (this.surfaceId === s.surfaceId) { this.surfaceId = ''; }
      }
    } as ISurfaceManagerListener);

    this.engine = new JankScenarioEngine(
      context,
      this.sm,
      (name: string, running: boolean, err: string | null) => {
        this.activeScenario = name;
        this.isRunning = running;
        this.errorText = err ?? '';
      },
      (pos: number) => { this.animPos = pos; }
    );
    this.engine.loadBaseline();
  }

  aboutToDisappear() {
    this.stopEverything();
    if (this.sm) { this.sm.destroy(); this.sm = null; }
    this.engine = null;
  }

  private stopEverything() {
    this.engine?.stop();
    this.smoothRunning = false;
  }

  private runScenario(name: string) {
    this.smoothRunning = false;
    this.animPos = 0;
    this.engine?.run(name);
  }

  private startSmoothAnim() {
    this.stopEverything();
    this.smoothRunning = true;
    this.animPos = 0;
    this.getUIContext().animateTo({ duration: 2000, curve: Curve.EaseInOut, iterations: -1, playMode: PlayMode.Alternate }, () => {
      this.animPos = 200;
    });
  }

  build() {
    Stack({ alignContent: Alignment.TopStart }) {
      Column() {
        // Top bar
        Row() {
          Text('‹')
            .fontSize(22).fontColor(Color.White).margin({ left: 16 })
            .onClick(() => router.back())
          Text('Jank Lab')
            .fontSize(18).fontWeight(FontWeight.Medium).fontColor(Color.White)
            .margin({ left: 8 }).layoutWeight(1)
        }
        .width('100%').height(56).backgroundColor('#6200EE').alignItems(VerticalAlign.Center)

        // Visible baseline surface (the "canvas")
        Scroll() {
          Column() {
            // ArkUI moving square (for SetTimeoutAnimation / smooth compare)
            Row() {
              Column()
                .width(40).height(40)
                .backgroundColor('#BB86FC')
                .margin({ left: this.animPos })
            }
            .width('100%').height(60)

            ForEach(
              this.surfaceId ? [this.surfaceId] : [],
              (sid: string) => {
                AGenUIContainer({ surfaceId: sid, instanceId: this.sm?.getInstanceId() ?? 0 })
                  .width('100%')
              },
              (_sid: string) => `jank_${this.surfaceVersion}`
            )
          }
          .width('100%')
        }
        .width('100%').layoutWeight(1).scrollBar(BarState.Off).edgeEffect(EdgeEffect.None)
        .backgroundColor(Color.White)

        // Scenario buttons
        Scroll() {
          Row() {
            ForEach(JankScenarioEngine.SCENARIO_NAMES, (name: string, i: number) => {
              Button(name)
                .fontSize(11)
                .backgroundColor(this.activeScenario === name && this.isRunning ? '#03DAC5' : COLORS[i % COLORS.length])
                .fontColor(Color.White)
                .margin({ left: 6, right: 6 })
                .onClick(() => this.runScenario(name))
            })
            Button('Stop')
              .fontSize(11).backgroundColor('#FF6666').fontColor(Color.White).margin({ left: 6, right: 6 })
              .onClick(() => this.stopEverything())
            Button('animateTo')
              .fontSize(11).backgroundColor('#4D95E5').fontColor(Color.White).margin({ left: 6, right: 6 })
              .onClick(() => this.startSmoothAnim())
          }
          .padding(8)
        }
        .width('100%').scrollBar(BarState.Off).align(Alignment.Center)
      }
      .width('100%').height('100%')

      // FPS overlay (top-left)
      FpsOverlay({ scenario: this.activeScenario, errorText: this.errorText })
    }
    .width('100%').height('100%')
  }
}
```
